# DaysOfWeekGet/__init__.py
import json
import os
import logging
import requests
import azure.functions as func

GRAPH_ROOT = "https://graph.microsoft.com/v1.0"
SITE_ID = "integracare.sharepoint.com,1d0ae5f6-8478-4c7f-a38d-6f72af2cb34f,478b6448-3c9d-4800-b8a2-a3bd74605d0d"


def _get_env(n: str) -> str:
    v = os.environ.get(n)
    if not v:
        raise Exception(f"Missing env var: {n}")
    return v


def _get_token() -> str:
    token_url = f"https://login.microsoftonline.com/{_get_env('AppRegTenantID')}/oauth2/v2.0/token"
    data = {
        "client_id": _get_env("AppRegClientID"),
        "client_secret": _get_env("AppRegSecretValue"),
        "grant_type": "client_credentials",
        "scope": "https://graph.microsoft.com/.default",
    }
    r = requests.post(token_url, data=data, timeout=30)
    r.raise_for_status()
    return r.json()["access_token"]


def _headers(tok: str): return {
    "Authorization": f"Bearer {tok}", "Accept": "application/json"}


def main(req: func.HttpRequest) -> func.HttpResponse:
    role = (req.route_params.get("role") or "").strip()
    if not role:
        return func.HttpResponse(json.dumps({"error": "role required"}), status_code=400, mimetype="application/json")

    try:
        tok = _get_token()
        # Path-based addressing to Shared Documents/DayInLifeCalendars/<role>.json
        url = f"{GRAPH_ROOT}/sites/{SITE_ID}/drive/root:/Shared Documents/DayInLifeCalendars/{role}.json:/content"
        r = requests.get(url, headers=_headers(tok), timeout=30)
        if r.status_code == 404:
            return func.HttpResponse(json.dumps({"version": 1}), status_code=200, mimetype="application/json")
        r.raise_for_status()
        return func.HttpResponse(r.text, status_code=200, mimetype="application/json")
    except Exception as e:
        logging.exception("DaysOfWeekGet failed")
        return func.HttpResponse(json.dumps({"error": str(e)}), status_code=500, mimetype="application/json")
